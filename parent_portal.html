async function searchStudent(){ const keyword=document .getElementById("parentSearch") .value .trim(); if(!keyword)return; document.getElementById("parentError").innerHTML=""; let student=null; const tables=[ "attendance_first_secondary", "attendance_second_secondary", "attendance_third_secondary" ]; for(const table of tables){ const {data}=await supabase .from(table) .select("*") .or(`student_id.eq.${keyword},qr_code.eq.${keyword}`) .limit(1); if(data && data.length){ student=data[0]; break; } } if(!student){ document.getElementById("parentError").innerHTML= "Student Not Found"; return; } window.currentStudent=student; document.getElementById("parentLogin").style.display="none"; loadParentDashboard(student); }
